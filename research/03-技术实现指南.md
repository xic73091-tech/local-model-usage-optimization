```python
# src/core/hardware_detector.py

import platform
import subprocess
from dataclasses import dataclass
from typing import Optional, List
import psutil

@dataclass
class GPUInfo:
    """GPU信息"""
    name: str
    vendor: str  # nvidia, amd, apple, intel
    vram_total: int  # MB
    vram_free: int  # MB
    cuda_version: Optional[str] = None
    compute_capability: Optional[str] = None

@dataclass
class CPUInfo:
    """CPU信息"""
    name: str
    cores_physical: int
    cores_logical: int
    frequency_mhz: float
    features: List[str]  # avx, avx2, avx512, etc.

@dataclass
class MemoryInfo:
    """内存信息"""
    total_gb: float
    available_gb: float
    swap_total_gb: float

@dataclass
class HardwareProfile:
    """硬件配置文件"""
    gpu: Optional[GPUInfo]
    cpu: CPUInfo
    memory: MemoryInfo
    os: str
    platform: str

class HardwareDetector:
    """硬件检测器"""

    def detect(self) -> HardwareProfile:
        """检测系统硬件配置"""
        return HardwareProfile(
            gpu=self._detect_gpu(),
            cpu=self._detect_cpu(),
            memory=self._detect_memory(),
            os=platform.system(),
            platform=platform.machine()
        )

    def _detect_gpu(self) -> Optional[GPUInfo]:
        """检测GPU信息"""
        # 尝试NVIDIA GPU
        try:
            result = subprocess.run(
                ['nvidia-smi', '--query-gpu=name,memory.total,memory.free,driver_version',
                 '--format=csv,noheader,nounits'],
                capture_output=True, text=True
            )
            if result.returncode == 0:
                parts = result.stdout.strip().split(', ')
                return GPUInfo(
                    name=parts[0],
                    vendor='nvidia',
                    vram_total=int(parts[1]),
                    vram_free=int(parts[2]),
                    cuda_version=self._get_cuda_version()
                )
        except FileNotFoundError:
            pass

        # 尝试AMD GPU (通过rocm-smi)
        try:
            result = subprocess.run(
                ['rocm-smi', '--showmeminfo', 'vram'],
                capture_output=True, text=True
            )
            if result.returncode == 0:
                # 解析AMD GPU信息
                return self._parse_amd_gpu_info(result.stdout)
        except FileNotFoundError:
            pass

        # Apple Silicon检测
        if platform.system() == 'Darwin' and platform.machine() == 'arm64':
            return self._detect_apple_gpu()

        return None

    def _detect_cpu(self) -> CPUInfo:
        """检测CPU信息"""
        cpu_info = CPUInfo(
            name=platform.processor() or "Unknown",
            cores_physical=psutil.cpu_count(logical=False) or 1,
            cores_logical=psutil.cpu_count(logical=True) or 1,
            frequency_mhz=psutil.cpu_freq().current if psutil.cpu_freq() else 0,
            features=self._detect_cpu_features()
        )
        return cpu_info

    def _detect_memory(self) -> MemoryInfo:
        """检测内存信息"""
        mem = psutil.virtual_memory()
        swap = psutil.swap_memory()
        return MemoryInfo(
            total_gb=mem.total / (1024**3),
            available_gb=mem.available / (1024**3),
            swap_total_gb=swap.total / (1024**3)
        )

    def _detect_cpu_features(self) -> List[str]:
        """检测CPU特性"""
        features = []
        if platform.system() == 'Linux':
            try:
                with open('/proc/cpuinfo', 'r') as f:
                    for line in f:
                        if line.startswith('flags'):
                            features = line.split(':')[1].strip().split()
                            break
            except:
                pass
        elif platform.system() == 'Windows':
            # Windows下通过CPUID检测
            features = self._detect_windows_cpu_features()
        return features

    def recommend_config(self, profile: HardwareProfile) -> 'ModelConfig':
        """基于硬件推荐最优配置"""
        from .config import ModelConfig

        config = ModelConfig()

        # GPU配置
        if profile.gpu:
            if profile.gpu.vram_total >= 24000:  # 24GB+
                config.max_model_size = "70B"
                config.quantization = "Q4_K_M"
                config.gpu_layers = -1  # 全部在GPU
            elif profile.gpu.vram_total >= 12000:  # 12GB+
                config.max_model_size = "13B"
                config.quantization = "Q4_K_M"
                config.gpu_layers = -1
            elif profile.gpu.vram_total >= 8000:  # 8GB+
                config.max_model_size = "7B"
                config.quantization = "Q4_K_S"
                config.gpu_layers = -1
            else:
                config.max_model_size = "3B"
                config.quantization = "Q4_K_S"
                config.gpu_layers = 20  # 部分在GPU
        else:
            # CPU推理
            config.max_model_size = "7B"
            config.quantization = "Q4_K_S"
            config.gpu_layers = 0
            config.threads = profile.cpu.cores_logical

        # 内存配置
        if profile.memory.total_gb >= 32:
            config.context_length = 8192
        elif profile.memory.total_gb >= 16:
            config.context_length = 4096
        else:
            config.context_length = 2048

        return config
```

### 2.2 模型管理模块

```python
# src/core/model_manager.py

import os
import json
from pathlib import Path
from typing import List, Optional, Dict, Any
from dataclasses import dataclass
from enum import Enum

class ModelFormat(Enum):
    GGUF = "gguf"
    SAFETENSORS = "safetensors"
    PYTORCH = "pytorch"
    ONNX = "onnx"
    AWQ = "awq"
    GPTQ = "gptq"

@dataclass
class ModelInfo:
    """模型信息"""
    id: str
    name: str
    path: Path
    format: ModelFormat
    size_gb: float
    parameters: str  # "7B", "13B", etc.
    quantization: Optional[str] = None
    metadata: Dict[str, Any] = None

class ModelManager:
    """模型管理器"""

    def __init__(self, models_dir: Path):
        self.models_dir = models_dir
        self.models_dir.mkdir(parents=True, exist_ok=True)
        self._index_file = models_dir / "index.json"
        self._load_index()

    def _load_index(self):
        """加载模型索引"""
        if self._index_file.exists():
            with open(self._index_file, 'r', encoding='utf-8') as f:
                self._index = json.load(f)
        else:
            self._index = {"models": {}, "last_scan": None}

    def _save_index(self):
        """保存模型索引"""
        with open(self._index_file, 'w', encoding='utf-8') as f:
            json.dump(self._index, f, indent=2, ensure_ascii=False)

    def scan_local_models(self) -> List[ModelInfo]:
        """扫描本地模型文件"""
        models = []

        # 扫描GGUF文件
        for gguf_file in self.models_dir.rglob("*.gguf"):
            model = self._analyze_gguf(gguf_file)
            if model:
                models.append(model)

        # 扫描SafeTensors文件
        for st_file in self.models_dir.rglob("*.safetensors"):
            model = self._analyze_safetensors(st_file)
            if model:
                models.append(model)

        # 扫描ONNX文件
        for onnx_file in self.models_dir.rglob("*.onnx"):
            model = self._analyze_onnx(onnx_file)
            if model:
                models.append(model)

        # 更新索引
        for model in models:
            self._index["models"][model.id] = {
                "name": model.name,
                "path": str(model.path),
                "format": model.format.value,
                "size_gb": model.size_gb,
                "parameters": model.parameters,
                "quantization": model.quantization
            }

        self._save_index()
        return models

    def _analyze_gguf(self, path: Path) -> Optional[ModelInfo]:
        """分析GGUF文件"""
        try:
            # 从文件名解析信息
            filename = path.stem
            parts = filename.split('-')

            # 尝试解析参数量和量化
            parameters = None
            quantization = None

            for part in parts:
                if part.endswith('B') or part.endswith('b'):
                    parameters = part.upper()
                elif part.startswith('Q') or part.startswith('q'):
                    quantization = part.upper()

            # 如果无法从文件名解析，尝试从GGUF头部读取
            if not parameters:
                parameters = self._extract_gguf_params(path)

            size_gb = path.stat().st_size / (1024**3)

            return ModelInfo(
                id=f"gguf_{filename}",
                name=filename,
                path=path,
                format=ModelFormat.GGUF,
                size_gb=size_gb,
                parameters=parameters or "Unknown",
                quantization=quantization
            )
        except Exception as e:
            print(f"Error analyzing GGUF {path}: {e}")
            return None

    def _extract_gguf_params(self, path: Path) -> Optional[str]:
        """从GGUF文件头部提取参数信息"""
        try:
            with open(path, 'rb') as f:
                # GGUF魔数
                magic = f.read(4)
                if magic != b'GGUF':
                    return None

                # 读取版本
                version = int.from_bytes(f.read(4), 'little')

                # 读取tensor数量
                tensor_count = int.from_bytes(f.read(8), 'little')

                # 读取metadata kv数量
                metadata_kv_count = int.from_bytes(f.read(8), 'little')

                # 解析metadata
                for _ in range(metadata_kv_count):
                    key = self._read_gguf_string(f)
                    value_type = int.from_bytes(f.read(4), 'little')

                    # 查找参数相关的key
                    if 'params' in key.lower() or 'size' in key.lower():
                        value = self._read_gguf_value(f, value_type)
                        if isinstance(value, (int, float)):
                            if value > 1e9:
                                return f"{value/1e9:.0f}B"
                            elif value > 1e6:
                                return f"{value/1e6:.0f}M"
                    else:
                        self._skip_gguf_value(f, value_type)

        except Exception:
            pass
        return None

    def download_model(self, model_id: str, source: str = "huggingface") -> Path:
        """从HuggingFace下载模型"""
        from huggingface_hub import hf_hub_download, snapshot_download

        try:
            # 解析model_id (格式: "org/model-name")
            if '/' in model_id:
                repo_id = model_id
            else:
                # 搜索模型
                from huggingface_hub import HfApi
                api = HfApi()
                models = api.list_models(search=model_id, limit=1)
                if models:
                    repo_id = models[0].id
                else:
                    raise ValueError(f"Model not found: {model_id}")

            # 下载模型
            local_dir = self.models_dir / repo_id.split('/')[-1]
            snapshot_download(
                repo_id=repo_id,
                local_dir=local_dir,
                local_dir_use_symlinks=False
            )

            print(f"Model downloaded to: {local_dir}")
            return local_dir

        except Exception as e:
            print(f"Error downloading model: {e}")
            raise

    def convert_format(
        self,
        model: ModelInfo,
        target_format: ModelFormat,
        quantization: Optional[str] = None
    ) -> Path:
        """转换模型格式"""
        if model.format == target_format:
            return model.path

        if target_format == ModelFormat.GGUF:
            return self._convert_to_gguf(model, quantization)
        elif target_format == ModelFormat.ONNX:
            return self._convert_to_onnx(model)
        else:
            raise ValueError(f"Unsupported conversion: {model.format} -> {target_format}")

    def _convert_to_gguf(self, model: ModelInfo, quantization: Optional[str]) -> Path:
        """转换为GGUF格式"""
        # 使用llama.cpp的convert脚本
        output_path = model.path.parent / f"{model.path.stem}.gguf"

        if model.format == ModelFormat.PYTORCH:
            # PyTorch -> GGUF
            import subprocess
            cmd = [
                'python', '-m', 'llama_cpp.convert',
                str(model.path),
                '--outfile', str(output_path)
            ]
            if quantization:
                cmd.extend(['--outtype', quantization])

            subprocess.run(cmd, check=True)

        return output_path

    def get_model_recommendation(self, task: str, hardware: 'HardwareProfile') -> Optional[ModelInfo]:
        """根据任务和硬件推荐模型"""
        # 任务到模型类型的映射
        task_model_map = {
            "chat": ["llama", "qwen", "mistral", "phi"],
            "code": ["codellama", "deepseek-coder", "starcoder"],
            "creative": ["llama", "mistral", "mixtral"],
            "analysis": ["llama", "qwen", "yi"],
            "translation": ["nllb", "madlad"],
        }

        # 获取推荐的模型族
        model_families = task_model_map.get(task, ["llama", "qwen"])

        # 根据硬件限制筛选
        max_size = "7B"
        if hardware.gpu and hardware.gpu.vram_total >= 24000:
            max_size = "70B"
        elif hardware.gpu and hardware.gpu.vram_total >= 12000:
            max_size = "13B"

        # 在本地模型中查找匹配的
        for model in self._index.get("models", {}).values():
            if any(family in model["name"].lower() for family in model_families):
                if self._size_fits(model["parameters"], max_size):
                    return model

        return None

    def _size_fits(self, model_size: str, max_size: str) -> bool:
        """检查模型大小是否在限制内"""
        def parse_size(s: str) -> int:
            s = s.upper().replace('B', '').replace('M', '')
            if 'M' in s.upper():
                return int(float(s) * 1e6)
            return int(float(s) * 1e9)

        try:
            return parse_size(model_size) <= parse_size(max_size)
        except:
            return True
```

```python
# src/backends/base.py

from abc import ABC, abstractmethod
from typing import List, Optional, AsyncIterator
from dataclasses import dataclass
from enum import Enum

class BackendType(Enum):
    LLAMA_CPP = "llama_cpp"
    VLLM = "vllm"
    ONNX_RUNTIME = "onnx_runtime"
    OPENVINO = "openvino"

@dataclass
class InferenceConfig:
    """推理配置"""
    model_path: str
    context_length: int = 4096
    gpu_layers: int = -1
    threads: int = 4
    batch_size: int = 512
    temperature: float = 0.7
    top_p: float = 0.9
    top_k: int = 40
    repeat_penalty: float = 1.1
    mlock: bool = True  # 锁定内存

@dataclass
class GenerationResult:
    """生成结果"""
    text: str
    tokens_generated: int
    tokens_per_second: float
    time_to_first_token: float
    total_time: float
    finish_reason: str  # "stop", "length", "error"

class InferenceBackend(ABC):
    """推理后端基类"""

    def __init__(self, config: InferenceConfig):
        self.config = config
        self._model = None
        self._is_loaded = False

    @abstractmethod
    async def load_model(self) -> bool:
        """加载模型"""
        pass

    @abstractmethod
    async def unload_model(self):
        """卸载模型"""
        pass

    @abstractmethod
    async def generate(
        self,
        prompt: str,
        max_tokens: int = 512,
        stop: Optional[List[str]] = None,
        stream: bool = False
    ) -> GenerationResult:
        """生成文本"""
        pass

    @abstractmethod
    async def generate_stream(
        self,
        prompt: str,
        max_tokens: int = 512,
        stop: Optional[List[str]] = None
    ) -> AsyncIterator[str]:
        """流式生成"""
        pass

    @abstractmethod
    async def get_embeddings(self, text: str) -> List[float]:
        """获取文本嵌入"""
        pass

    @abstractmethod
    def get_memory_usage(self) -> dict:
        """获取内存使用情况"""
        pass

    @property
    def is_loaded(self) -> bool:
        return self._is_loaded
```

```python
# src/backends/llama_cpp.py

import asyncio
from typing import List, Optional, AsyncIterator
from llama_cpp import Llama, LlamaGrammar
from .base import InferenceBackend, InferenceConfig, GenerationResult
import time

class LlamaCppBackend(InferenceBackend):
    """llama.cpp推理后端"""

    def __init__(self, config: InferenceConfig):
        super().__init__(config)
        self._model: Optional[Llama] = None

    async def load_model(self) -> bool:
        """加载模型"""
        try:
            self._model = Llama(
                model_path=self.config.model_path,
                n_ctx=self.config.context_length,
                n_gpu_layers=self.config.gpu_layers,
                n_threads=self.config.threads,
                n_batch=self.config.batch_size,
                use_mlock=self.config.mlock,
                verbose=False
            )
            self._is_loaded = True
            return True
        except Exception as e:
            print(f"Error loading model: {e}")
            return False

    async def unload_model(self):
        """卸载模型"""
        if self._model:
            del self._model
            self._model = None
            self._is_loaded = False
            import gc
            gc.collect()

    async def generate(
        self,
        prompt: str,
        max_tokens: int = 512,
        stop: Optional[List[str]] = None,
        stream: bool = False
    ) -> GenerationResult:
        """生成文本"""
        if not self._is_loaded:
            raise RuntimeError("Model not loaded")

        start_time = time.time()
        ttft = None

        # 创建生成器
        generator = self._model(
            prompt,
            max_tokens=max_tokens,
            stop=stop,
            temperature=self.config.temperature,
            top_p=self.config.top_p,
            top_k=self.config.top_k,
            repeat_penalty=self.config.repeat_penalty,
            stream=True
        )

        # 收集生成的文本
        generated_text = ""
        tokens_generated = 0

        for chunk in generator:
            if ttft is None:
                ttft = time.time() - start_time

            token = chunk['choices'][0]['text']
            generated_text += token
            tokens_generated += 1

        total_time = time.time() - start_time
        tokens_per_second = tokens_generated / total_time if total_time > 0 else 0

        return GenerationResult(
            text=generated_text,
            tokens_generated=tokens_generated,
            tokens_per_second=tokens_per_second,
            time_to_first_token=ttft or 0,
            total_time=total_time,
            finish_reason="stop"
        )

    async def generate_stream(
        self,
        prompt: str,
        max_tokens: int = 512,
        stop: Optional[List[str]] = None
    ) -> AsyncIterator[str]:
        """流式生成"""
        if not self._is_loaded:
            raise RuntimeError("Model not loaded")

        generator = self._model(
            prompt,
            max_tokens=max_tokens,
            stop=stop,
            temperature=self.config.temperature,
            top_p=self.config.top_p,
            top_k=self.config.top_k,
            repeat_penalty=self.config.repeat_penalty,
            stream=True
        )

        for chunk in generator:
            token = chunk['choices'][0]['text']
            yield token

    async def get_embeddings(self, text: str) -> List[float]:
        """获取文本嵌入"""
        if not self._is_loaded:
            raise RuntimeError("Model not loaded")

        embedding = self._model.embed(text)
        return embedding.tolist()

    def get_memory_usage(self) -> dict:
        """获取内存使用情况"""
        import psutil
        process = psutil.Process()

        return {
            "rss_mb": process.memory_info().rss / 1024 / 1024,
            "vms_mb": process.memory_info().vms / 1024 / 1024,
            "gpu_mb": self._get_gpu_memory() if self.config.gpu_layers > 0 else 0
        }

    def _get_gpu_memory(self) -> float:
        """获取GPU显存使用"""
        try:
            import subprocess
            result = subprocess.run(
                ['nvidia-smi', '--query-gpu=memory.used', '--format=csv,noheader,nounits'],
                capture_output=True, text=True
            )
            if result.returncode == 0:
                return float(result.stdout.strip())
        except:
            pass
        return 0.0
```

```python
# src/optimization/scheduler.py

import asyncio
from typing import List, Dict, Optional, Callable
from dataclasses import dataclass
from enum import Enum
import heapq
from collections import defaultdict

class TaskType(Enum):
    CHAT = "chat"
    CODE = "code"
    CREATIVE = "creative"
    ANALYSIS = "analysis"
    TRANSLATION = "translation"
    EMBEDDING = "embedding"

class Priority(Enum):
    LOW = 0
    NORMAL = 1
    HIGH = 2
    URGENT = 3

@dataclass
class InferenceRequest:
    """推理请求"""
    id: str
    task_type: TaskType
    prompt: str
    max_tokens: int
    priority: Priority
    callback: Optional[Callable] = None
    metadata: Dict = None

    def __lt__(self, other):
        return self.priority.value > other.priority.value

@dataclass
class ModelProfile:
    """模型性能画像"""
    model_id: str
    task_scores: Dict[TaskType, float]  # 任务类型 -> 性能得分
    tokens_per_second: float
    memory_usage_mb: float
    load_time_seconds: float

class InferenceScheduler:
    """智能推理调度器"""

    def __init__(self, max_concurrent: int = 3):
        self.max_concurrent = max_concurrent
        self._request_queue: List[InferenceRequest] = []
        self._active_requests: Dict[str, asyncio.Task] = {}
        self._model_profiles: Dict[str, ModelProfile] = {}
        self._loaded_models: Dict[str, any] = {}
        self._lock = asyncio.Lock()

        # 性能统计
        self._stats = defaultdict(lambda: {
            "total_requests": 0,
            "total_tokens": 0,
            "total_time": 0,
            "errors": 0
        })

    def register_model(self, profile: ModelProfile):
        """注册模型性能画像"""
        self._model_profiles[profile.model_id] = profile

    async def submit_request(self, request: InferenceRequest) -> asyncio.Future:
        """提交推理请求"""
        async with self._lock:
            heapq.heappush(self._request_queue, request)

        # 创建Future用于返回结果
        future = asyncio.Future()

        # 如果有空闲槽位，立即处理
        if len(self._active_requests) < self.max_concurrent:
            asyncio.create_task(self._process_next())

        return future

    async def _process_next(self):
        """处理下一个请求"""
        async with self._lock:
            if not self._request_queue:
                return

            request = heapq.heappop(self._request_queue)

        # 选择最优模型
        model_id = await self._select_model(request)

        # 执行推理
        task = asyncio.create_task(
            self._execute_inference(request, model_id)
        )
        self._active_requests[request.id] = task

        try:
            result = await task
            if request.callback:
                request.callback(result)
        finally:
            del self._active_requests[request.id]
            # 继续处理下一个
            if self._request_queue:
                asyncio.create_task(self._process_next())

    async def _select_model(self, request: InferenceRequest) -> str:
        """根据请求选择最优模型"""
        if not self._model_profiles:
            raise ValueError("No models registered")

        best_model = None
        best_score = -1

        for model_id, profile in self._model_profiles.items():
            # 计算综合得分
            score = self._calculate_model_score(profile, request)

            if score > best_score:
                best_score = score
                best_model = model_id

        return best_model

    def _calculate_model_score(
        self,
        profile: ModelProfile,
        request: InferenceRequest
    ) -> float:
        """计算模型匹配得分"""
        # 任务匹配得分 (0-1)
        task_score = profile.task_scores.get(request.task_type, 0.5)

        # 速度得分 (归一化到0-1)
        speed_score = min(profile.tokens_per_second / 100, 1.0)

        # 内存效率得分 (越低越好)
        memory_score = 1.0 - min(profile.memory_usage_mb / 16000, 1.0)

        # 加权综合
        weights = {
            TaskType.CHAT: {"task": 0.5, "speed": 0.3, "memory": 0.2},
            TaskType.CODE: {"task": 0.6, "speed": 0.2, "memory": 0.2},
            TaskType.CREATIVE: {"task": 0.4, "speed": 0.2, "memory": 0.4},
            TaskType.ANALYSIS: {"task": 0.5, "speed": 0.2, "memory": 0.3},
        }

        w = weights.get(request.task_type, weights[TaskType.CHAT])

        return (
            task_score * w["task"] +
            speed_score * w["speed"] +
            memory_score * w["memory"]
        )

    async def _execute_inference(
        self,
        request: InferenceRequest,
        model_id: str
    ):
        """执行推理"""
        import time
        start_time = time.time()

        try:
            # 获取或加载模型
            backend = await self._get_or_load_model(model_id)

            # 执行推理
            result = await backend.generate(
                prompt=request.prompt,
                max_tokens=request.max_tokens
            )

            # 更新统计
            self._stats[model_id]["total_requests"] += 1
            self._stats[model_id]["total_tokens"] += result.tokens_generated
            self._stats[model_id]["total_time"] += result.total_time

            return result

        except Exception as e:
            self._stats[model_id]["errors"] += 1
            raise

    async def _get_or_load_model(self, model_id: str):
        """获取或加载模型"""
        if model_id not in self._loaded_models:
            # 这里需要根据model_id创建对应的backend
            # 简化示例
            from ..backends.llama_cpp import LlamaCppBackend, InferenceConfig

            config = InferenceConfig(
                model_path=self._get_model_path(model_id)
            )
            backend = LlamaCppBackend(config)
            await backend.load_model()
            self._loaded_models[model_id] = backend

        return self._loaded_models[model_id]

    def get_stats(self) -> Dict:
        """获取调度统计"""
        return dict(self._stats)
```

```python
# src/monitor/metrics.py

import time
import psutil
import asyncio
from typing import Dict, List, Optional
from dataclasses import dataclass, field
from collections import deque
import threading

@dataclass
class PerformanceMetrics:
    """性能指标"""
    timestamp: float

    # 推理指标
    tokens_per_second: float = 0
    time_to_first_token: float = 0
    total_requests: int = 0
    active_requests: int = 0
    queue_size: int = 0

    # 资源指标
    cpu_percent: float = 0
    memory_percent: float = 0
    memory_used_gb: float = 0
    gpu_percent: float = 0
    gpu_memory_used_gb: float = 0
    gpu_temperature: float = 0

    # 缓存指标
    cache_hit_rate: float = 0
    cache_size_mb: float = 0

class MetricsCollector:
    """指标收集器"""

    def __init__(self, history_size: int = 1000):
        self._history: deque = deque(maxlen=history_size)
        self._lock = threading.Lock()
        self._running = False
        self._collection_interval = 1.0  # 秒

    async def start(self):
        """启动指标收集"""
        self._running = True
        while self._running:
            metrics = await self._collect_metrics()
            with self._lock:
                self._history.append(metrics)
            await asyncio.sleep(self._collection_interval)

    def stop(self):
        """停止指标收集"""
        self._running = False

    async def _collect_metrics(self) -> PerformanceMetrics:
        """收集当前指标"""
        metrics = PerformanceMetrics(timestamp=time.time())

        # CPU指标
        metrics.cpu_percent = psutil.cpu_percent(interval=0.1)
        mem = psutil.virtual_memory()
        metrics.memory_percent = mem.percent
        metrics.memory_used_gb = mem.used / (1024**3)

        # GPU指标
        gpu_info = await self._get_gpu_metrics()
        if gpu_info:
            metrics.gpu_percent = gpu_info.get("utilization", 0)
            metrics.gpu_memory_used_gb = gpu_info.get("memory_used", 0)
            metrics.gpu_temperature = gpu_info.get("temperature", 0)

        return metrics

    async def _get_gpu_metrics(self) -> Optional[Dict]:
        """获取GPU指标"""
        try:
            import subprocess
            result = subprocess.run(
                ['nvidia-smi', '--query-gpu=utilization.gpu,memory.used,temperature.gpu',
                 '--format=csv,noheader,nounits'],
                capture_output=True, text=True
            )
            if result.returncode == 0:
                parts = result.stdout.strip().split(', ')
                return {
                    "utilization": float(parts[0]),
                    "memory_used": float(parts[1]) / 1024,  # MB to GB
                    "temperature": float(parts[2])
                }
        except:
            pass
        return None

    def get_current_metrics(self) -> Optional[PerformanceMetrics]:
        """获取最新指标"""
        with self._lock:
            return self._history[-1] if self._history else None

    def get_history(self, duration_seconds: int = 60) -> List[PerformanceMetrics]:
        """获取历史指标"""
        cutoff = time.time() - duration_seconds
        with self._lock:
            return [m for m in self._history if m.timestamp >= cutoff]

    def get_average_metrics(self, duration_seconds: int = 60) -> Optional[PerformanceMetrics]:
        """获取平均指标"""
        history = self.get_history(duration_seconds)
        if not history:
            return None

        avg = PerformanceMetrics(timestamp=time.time())
        count = len(history)

        avg.tokens_per_second = sum(m.tokens_per_second for m in history) / count
        avg.cpu_percent = sum(m.cpu_percent for m in history) / count
        avg.memory_percent = sum(m.memory_percent for m in history) / count
        avg.gpu_percent = sum(m.gpu_percent for m in history) / count

        return avg


class PerformanceAnalyzer:
    """性能分析器"""

    def __init__(self, collector: MetricsCollector):
        self.collector = collector

    def analyze_bottleneck(self) -> Dict:
        """分析性能瓶颈"""
        metrics = self.collector.get_average_metrics(60)
        if not metrics:
            return {"bottleneck": "unknown", "details": "No metrics available"}

        bottlenecks = []

        # CPU瓶颈
        if metrics.cpu_percent > 90:
            bottlenecks.append({
                "type": "cpu",
                "severity": "high",
                "value": metrics.cpu_percent,
                "suggestion": "CPU使用率过高，考虑减少CPU推理或增加GPU使用"
            })

        # 内存瓶颈
        if metrics.memory_percent > 90:
            bottlenecks.append({
                "type": "memory",
                "severity": "high",
                "value": metrics.memory_percent,
                "suggestion": "内存使用率过高，考虑使用更小的模型或增加量化"
            })

        # GPU瓶颈
        if metrics.gpu_percent > 95:
            bottlenecks.append({
                "type": "gpu",
                "severity": "high",
                "value": metrics.gpu_percent,
                "suggestion": "GPU使用率过高，考虑减少并发请求或使用更小的模型"
            })

        # GPU显存瓶颈
        if metrics.gpu_memory_used_gb > 0:
            # 假设总显存为8GB或更多
            gpu_memory_percent = (metrics.gpu_memory_used_gb / 8) * 100
            if gpu_memory_percent > 90:
                bottlenecks.append({
                    "type": "gpu_memory",
                    "severity": "high",
                    "value": gpu_memory_percent,
                    "suggestion": "GPU显存不足，考虑使用更小的模型或增加量化"
                })

        if not bottlenecks:
            return {"bottleneck": "none", "details": "系统运行正常"}

        # 返回最严重的瓶颈
        bottlenecks.sort(key=lambda x: {"high": 3, "medium": 2, "low": 1}[x["severity"]])
        return bottlenecks[0]

    def suggest_optimization(self) -> List[Dict]:
        """提供优化建议"""
        suggestions = []
        metrics = self.collector.get_average_metrics(60)

        if not metrics:
            return suggestions

        # 基于tokens/s的建议
        if metrics.tokens_per_second < 10:
            suggestions.append({
                "category": "speed",
                "priority": "high",
                "title": "推理速度较慢",
                "description": f"当前速度: {metrics.tokens_per_second:.1f} tokens/s",
                "actions": [
                    "尝试使用更小的模型",
                    "增加量化级别(如从Q5降到Q4)",
                    "减少上下文长度",
                    "启用GPU加速(如果可用)"
                ]
            })

        # 基于TTFT的建议
        if metrics.time_to_first_token > 1.0:
            suggestions.append({
                "category": "latency",
                "priority": "medium",
                "title": "首Token延迟较高",
                "description": f"当前延迟: {metrics.time_to_first_token:.2f}s",
                "actions": [
                    "减少提示词长度",
                    "使用更快的模型",
                    "启用KV Cache"
                ]
            })

        # 基于内存的建议
        if metrics.memory_percent > 80:
            suggestions.append({
                "category": "memory",
                "priority": "high",
                "title": "内存使用率较高",
                "description": f"当前使用率: {metrics.memory_percent:.1f}%",
                "actions": [
                    "使用更小的模型",
                    "增加量化级别",
                    "减少上下文长度",
                    "关闭其他占用内存的程序"
                ]
            })

        return suggestions
```
